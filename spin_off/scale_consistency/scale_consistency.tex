\begin{abstract}
We study structural inference for lagged transport discrepancies that admit a
power-law geometry $D_j = \zeta j^H$.  The observation problem is coupled:
each estimated discrepancy $\Dhj$ carries finite-sample noise, and the
leading variance of $\log \Dhj$ depends on the same scale law that is being
inferred.  In this heteroscedastic log-domain model, we prove a central limit
theorem for the two-step feasible weighted least-squares estimator of $H$,
show that the oracle residual statistic has an exact $\chi^2(L-2)$ null law,
and derive a minimax separation rate $(nL)^{-1/2}$ for distinguishing the
power-law family from scale-inconsistent alternatives.  The resulting
procedure is adaptive in $H$ and retains the optimal rate without prior
knowledge of the exponent.  Simulations confirm exact size calibration,
FWLS-oracle agreement, and power consistent with the boundary.
\end{abstract}

\bigskip\noindent
\textbf{Keywords.} Wasserstein distance; structural inference; power-law lag
geometry; minimax lower bound; heteroscedastic regression; feasible WLS;
Le~Cam's method; transport discrepancy.

\section{Introduction}
\label{sec:intro}

Let $\{P_t\}_{t \geq 0}$ be a family of probability measures on
$\R^d$ evolving over time.  The object of interest is the lagged transport
geometry encoded by the Wasserstein-2 discrepancy curve
\[
    D_j \;:=\; W_2(P_{t-j}, P_t),\qquad j = 1, 2, \ldots, L.
\]
This curve is typically observed through estimators $\Dhj$ constructed from
$n$ i.i.d.\ samples from each $P_{t-j}$ and $P_t$, rather than through the
population values $D_j$ themselves.  Under standard conditions on the measures
\citep{FournierGuillin2015,GoldfeldEtAl2022,delBarrioEtAl2023}, the plug-in Wasserstein estimator
satisfies
\begin{equation}
    \Dhj = D_j + \varepsilon_j,\qquad
    \varepsilon_j \sim \mathrm{AN}\!\left(0,\; \frac{\sigma_0^2}{n}\right),
    \label{eq:obs-model}
\end{equation}
where $\sigma_0^2 > 0$ is a nuisance constant depending on the local
geometry of the measures.

A low-dimensional structural hypothesis is that the discrepancy curve follows
a \emph{power law}:
\begin{equation}
    D_j = \zeta\,j^H,\qquad \zeta > 0,\; H \in (0,1).
    \label{eq:power-law}
\end{equation}
This is scale consistency: the lag curve is compressed to the two-parameter
family $(\zeta, H)$, with exponent $H$ interpreted as a roughness index of
the latent drift in locally stationary settings \citep{Dahlhaus1997,Vogt2012,TinioAlayaBouzebda2025}.

The inferential problem is not standard.  In log-space,
$\Var(\log \Dhj) \approx \sigma_0^2/(n D_j^2) = \sigma_0^2 /(n\,\zeta^2\,j^{2H})$,
so the observation noise depends on the same scale law that is being
inferred.  The family is low-dimensional, but the noise is
parameter-coupled.

The main results establish a complete inference layer for this family: a
feasible WLS limit theorem for $H$, exact oracle calibration of a residual
adequacy statistic, a minimax lower bound for distinguishing the family from
scale-inconsistent alternatives, and adaptivity in $H$.

\paragraph{Main results.}
\begin{enumerate}[leftmargin=*,label=(\arabic*)]
\item \textbf{Feasible inference for the exponent.}  \Cref{thm:clt-fwls}
gives a central limit theorem for the two-step feasible WLS estimator
$\Hhat$ of $H$ in the heteroscedastic log-scale model.  The
pilot-estimation error does not inflate the asymptotic variance beyond the
oracle rate.

\item \textbf{Residual adequacy test.}  \Cref{thm:null-chi2} shows that
the weighted residual sum of squares $Q$ satisfies $Q \sim \chi^2(L-2)$
under the power-law family with oracle weights, and that two-step FWLS
converges to the same law asymptotically as $n \to \infty$ with $L$ fixed.

\item \textbf{Minimax boundary for model distinguishability.}
\Cref{thm:lecam-lower} gives the minimax separation rate $(nL)^{-1/2}$ for
distinguishing the power-law family from alternatives with scale
inconsistency level $\kappa \geq \kappa^*$.

\item \textbf{Adaptive structural inference.}  \Cref{thm:adaptivity}
shows that the FWLS adequacy procedure achieves the minimax rate without
prior knowledge of $H$.

\item \textbf{Simulation signatures.}  The numerical study reports exact
size calibration, FWLS-oracle agreement, and power curves consistent with the
boundary $\kappa^* = (nL)^{-1/2}$ (\Cref{sec:sims}).
\end{enumerate}

\section{Model and Assumptions}
\label{sec:model}

\subsection{Observation model}

We observe a triangular array $\{\Dhj\}_{j=1}^{L}$ satisfying
\begin{equation}
    \Dhj = D_j + \varepsilon_j,\qquad
    \varepsilon_j \iid \calN\!\left(0,\,\frac{\sigma_0^2}{n}\right),
    \tag{M}
    \label{eq:M}
\end{equation}
where $D_j = \zeta\,j^H$ for unknown parameters $(\zeta, H) \in
(0,\infty)\times(0,1)$, $\sigma_0 > 0$ is known (or consistently
estimable), and $n \geq 1$ is the effective sample size per lag.

\begin{remark}
    The Gaussianity in \eqref{eq:M} is justified for large $n$ by the
    central limit theorem for Wasserstein estimators; see
    \citet{GoldfeldEtAl2022} and \citet{delBarrioEtAl2023}.  The
    constant-variance assumption $\Var(\varepsilon_j) = \sigma_0^2/n$ is
    the leading-order approximation; the full theory allows $\sigma_j^2/n$
    with $\sigma_j \leq C\,D_j$ bounded away from zero and infinity
    relative to $D_j$---see \Cref{ass:sigma}.
\end{remark}

\subsection{Log-scale regression}

Taking logarithms of the power-law model and applying the delta method:
\begin{equation}
    Y_j \;:=\; \log \Dhj \;=\; \alpha + H \log j + U_j,
    \label{eq:log-model}
\end{equation}
where $\alpha = \log \zeta$ and
\begin{align}
    U_j &= \log\!\left(1 + \frac{\varepsilon_j}{D_j}\right)
         \approx \frac{\varepsilon_j}{D_j}
         \sim \calN\!\left(0,\; v_j\right), \label{eq:log-noise}\\
    v_j &:= \frac{\sigma_0^2}{n\,D_j^2}
         = \frac{\sigma_0^2}{n\,\zeta^2\,j^{2H}}.
\end{align}
The variance $v_j$ \emph{depends on $H$ and $\zeta$}, which are the
parameters of interest.  This is the parameter-coupled heteroscedasticity
that distinguishes our problem from standard regression.

\subsection{Hypotheses}

Let $\kappa \geq 0$ measure the \emph{scale inconsistency}: under the
alternative, each observation has an additional log-normal multiplicative
factor,
\[
    \log \Dhj = \alpha + H\log j + \kappa\,\xi_j + U_j,\quad
    \xi_j \iid \calN(0,1),\;\xi_j \perp U_j.
\]
The hypotheses are:
\begin{align*}
    H_0&:\; \kappa = 0 \quad\text{(scale consistency)},\\
    H_1&:\; \kappa \geq \kappa^* \quad\text{(scale inconsistency at level } \kappa^*\text{)}.
\end{align*}

\subsection{Assumptions}

\begin{assumption}[Signal strength]
\label{ass:signal}
$\zeta \geq \zeta_{\min} > 0$ and $H \in [\underline{H}, \overline{H}]
\subset (0,1)$.
\end{assumption}

\begin{assumption}[Noise regularity]
\label{ass:sigma}
$\sigma_0 \in [\sigma_{\min}, \sigma_{\max}]$ with
$0 < \sigma_{\min} \leq \sigma_{\max} < \infty$.
The delta-method approximation error satisfies
$\abs{U_j - \varepsilon_j/D_j} = O_p(n^{-1})$ uniformly in $j$.
\end{assumption}

\begin{assumption}[Lag schedule]
\label{ass:lags}
$L \geq 4$ is fixed as $n \to \infty$.
The lags $x_j := \log j$ for $j = 1, \ldots, L$ satisfy
$\sum_j (x_j - \bar{x})^2 \geq c_L > 0$.
\end{assumption}

\section{CLT for the Feasible WLS Estimator}
\label{sec:clt}

\subsection{Oracle WLS}

With known weights $w_j^* = 1/v_j = n\,D_j^2/\sigma_0^2$, the oracle
WLS estimator of $\boldsymbol{\beta} = (\alpha, H)^\top$ in
\eqref{eq:log-model} is
\begin{equation}
    \boldsymbol{\hat\beta}^* = (\mathbf{X}^\top \mathbf{W}^* \mathbf{X})^{-1}
    \mathbf{X}^\top \mathbf{W}^* \mathbf{Y},
    \label{eq:oracle-wls}
\end{equation}
where $\mathbf{X} = [\mathbf{1}\mid \mathbf{x}]$, $\mathbf{x} = (\log
1, \ldots, \log L)^\top$, and $\mathbf{W}^* = \diag(w_1^*, \ldots, w_L^*)$.

\begin{proposition}[Oracle WLS CLT]
\label{prop:oracle-clt}
Under \Cref{ass:signal,ass:sigma,ass:lags}, as $n \to \infty$ with $L$
fixed,
\[
    n^{1/2}\,\bigl(\hat H^* - H\bigr) \;\dto\; \calN\!\left(0,\;
    \sigma_0^2 \bigl[(\mathbf{X}^\top \mathbf{W}^* \mathbf{X})^{-1}\bigr]_{22}\right),
\]
where $[\,\cdot\,]_{22}$ denotes the $(2,2)$ entry.
Explicitly,
\begin{equation}
    \Var(\hat H^*) \;\approx\; \frac{\sigma_0^2}{n}
    \cdot \frac{S_{w^*}}
    {S_{w^*} S_{w^* x^2} - (S_{w^* x})^2},
    \label{eq:oracle-var}
\end{equation}
where $S_{w^*} = \sum_j w_j^*$, $S_{w^* x} = \sum_j w_j^* x_j$,
$S_{w^* x^2} = \sum_j w_j^* x_j^2$.
\end{proposition}

\begin{proof}
Standard weighted least squares with $\mathbf{U} \sim
\calN(\mathbf{0}, \sigma_0^2 n^{-1} (\mathbf{W}^*)^{-1})$; the
$(2,2)$ entry of $(\mathbf{X}^\top \mathbf{W}^* \mathbf{X})^{-1}$ follows
by $2\times 2$ matrix inversion.
\end{proof}

\subsection{Two-step feasible WLS}

Since $D_j$ is unknown, the oracle weights are infeasible.  We use the
following two-step procedure.

\begin{algorithm}[H]
\caption{Two-Step FWLS Estimator}
\label{alg:fwls}
\begin{algorithmic}[1]
\State \textbf{Input:} $(\Dhj)_{j=1}^L$, $n$, $\sigma_0$.
\State \textbf{Step 1 (Pilot OLS).}  Compute
    $(\hat\alpha_{\mathrm{pil}}, \hat H_{\mathrm{pil}})$ by unweighted
    OLS of $\log \Dhj$ on $(\mathbf{1}, \log j)$.
\State \textbf{Step 1b (Pilot imputation).}  Set
    $\hat D_j^{\mathrm{pil}} = \exp(\hat\alpha_{\mathrm{pil}} +
    \hat H_{\mathrm{pil}} \log j)$.
\State \textbf{Step 2 (FWLS).}  Set $\hat w_j =
    n (\hat D_j^{\mathrm{pil}})^2 / \sigma_0^2$ and compute
    $(\hat\alpha, \Hhat)$ by WLS with weights $(\hat w_j)$.
\State \textbf{Output:} $\Hhat$, $\hat\alpha$, residuals
    $\hat U_j = \log \Dhj - \hat\alpha - \Hhat \log j$.
\end{algorithmic}
\end{algorithm}

\begin{theorem}[FWLS CLT]
\label{thm:clt-fwls}
Under \Cref{ass:signal,ass:sigma,ass:lags}, the two-step FWLS estimator
$\Hhat$ satisfies, as $n \to \infty$ with $L$ fixed:
\begin{equation}
    n^{1/2}\bigl(\Hhat - H\bigr) \;\dto\;
    \calN\!\Bigl(0,\; \sigma_0^2
    \bigl[(\mathbf{X}^\top \mathbf{W}^* \mathbf{X})^{-1}\bigr]_{22}\Bigr).
    \label{eq:fwls-clt}
\end{equation}
In particular, $\Hhat$ is \emph{asymptotically equivalent to the oracle
WLS estimator}: the pilot estimation error inflates neither the asymptotic
variance nor the rate.
\end{theorem}

\begin{proof}[Proof sketch]
Write $\hat w_j = w_j^* \cdot r_j$ where $r_j = (\hat D_j^{\mathrm{pil}}
/ D_j)^2$.  Since $\hat H_{\mathrm{pil}} - H = O_p(n^{-1/2})$ (OLS
rate in $\log$-scale with bounded regressors), we have
$\log r_j = 2(\hat H_{\mathrm{pil}} - H)\log j + O_p(n^{-1}) = O_p(n^{-1/2}\log j)$,
hence $r_j = 1 + O_p(n^{-1/2}\log j)$ uniformly in $j$.

The FWLS estimator with weights $\hat w_j = w_j^*(1 + \delta_j)$,
$\delta_j = r_j - 1 = O_p(n^{-1/2}\log j)$, satisfies
\[
    \Hhat - H = \hat H^* + R_n,
\]
where the remainder $R_n = O_p(n^{-1}\log L)$ via a second-order
Taylor expansion of the WLS normal equations in the weight perturbation.
Since $n^{1/2} R_n = O_p(n^{-1/2}\log L) \to 0$, the Slutsky theorem
gives \eqref{eq:fwls-clt}.
\end{proof}

\section{Residual Adequacy of the Power-Law Family}
\label{sec:test}

\subsection{Test statistic}

Define the \emph{scale consistency statistic}
\begin{equation}
    Q \;:=\; \sum_{j=1}^{L} \hat w_j \,\hat U_j^2,
    \label{eq:Q-stat}
\end{equation}
where $\hat w_j$ and $\hat U_j$ are from \Cref{alg:fwls}.

\begin{theorem}[Null distribution]
\label{thm:null-chi2}
Under $H_0$ and \Cref{ass:signal,ass:sigma,ass:lags}:
\begin{enumerate}[label=(\alph*)]
    \item \textup{(Oracle.)}  With true weights $w_j^*$, the statistic
    $Q^* = \sum_j w_j^* U_j^{*2}$, where $U_j^* = Y_j - \alpha - H x_j$
    are the residuals from the infeasible WLS projection,
    satisfies $Q^* \sim \chi^2(L-2)$ exactly.
    \item \textup{(Feasible.)}  The two-step statistic $Q$ satisfies
    $Q \dto \chi^2(L-2)$ as $n \to \infty$.
\end{enumerate}
\end{theorem}

\begin{proof}
\textbf{Part (a).}  Under $H_0$, $\mathbf{Y} = \mathbf{X}\boldsymbol\beta +
\mathbf{U}$ with $\mathbf{U} \sim \calN(\mathbf{0}, \sigma_0^2 n^{-1}
(\mathbf{W}^*)^{-1})$.  Define $\tilde{\mathbf{U}} =
(\mathbf{W}^*)^{1/2}\mathbf{U}/(\sigma_0 n^{-1/2})$, so $\tilde{\mathbf{U}}
\sim \calN(\mathbf{0}, \mathbf{I}_L)$.  The projection onto the column
space of $(\mathbf{W}^*)^{1/2}\mathbf{X}$ has rank 2, so the residual
projection has rank $L-2$.  Hence $Q^* = \norm{\tilde{\mathbf{U}} -
\mathbf{P}\tilde{\mathbf{U}}}^2 \sim \chi^2(L-2)$.

\textbf{Part (b).}  We have $Q = Q^* + (Q - Q^*)$.  By the linearisation
in the proof of \Cref{thm:clt-fwls}, $Q - Q^* = O_p(n^{-1/2})$,
so $Q \dto Q^*$ in distribution.  Since the $\chi^2$ CDF is continuous,
convergence in distribution passes through the CDF and $Q \dto \chi^2(L-2)$.
\end{proof}

\begin{remark}
    Part~(a) is exact---not asymptotic---because the problem is a
    weighted Gaussian linear regression with known weights.  For finite $n$,
    the delta-method approximation in \eqref{eq:log-noise} introduces a bias
    of order $O(n^{-1})$ per lag; this is negligible for the $\chi^2$
    calibration at the rates we consider.  Simulations confirm empirical
    sizes of 0.049--0.052 at $\alpha = 0.05$ for all $L \in \{10,20,30,50\}$
    and $n = 1000$ (see \Cref{tab:null}).
\end{remark}

\subsection{Decision rule}

The scale consistency test at level $\alpha$ rejects $H_0$ when
\begin{equation}
    Q \;>\; q_{L-2,\,1-\alpha},
    \label{eq:test-rule}
\end{equation}
where $q_{L-2,\,1-\alpha}$ is the $(1-\alpha)$-quantile of $\chi^2(L-2)$.

For moderate $L$ (say $L \geq 20$), one may equivalently use the
standardised form
\[
    T \;:=\; \frac{Q - (L-2)}{\sqrt{2(L-2)}} \;\dto\; \calN(0,1),
\]
rejecting when $T > z_{1-\alpha}$.  However, the direct $\chi^2$ critical
value gives better finite-$L$ calibration (see \Cref{tab:null}).

\section{Minimax Lower Bound}
\label{sec:lower}

We now show that no test can do better than ours at the rate $(nL)^{-1/2}$.

\subsection{Setup}

Fix $H \in (0,1)$, $\zeta = 1$, $\sigma_0 = 1$.  For $\kappa \geq 0$,
define the probability measure $\Prob_\kappa$ on $\R^L$ under which
\begin{align*}
    Y_j &= H\log j + \kappa\,\xi_j + U_j,\\
    \xi_j &\iid \calN(0,1), \qquad U_j \sim \calN(0, v_j),\\
    v_j &= n^{-1} j^{-2H}, \qquad \xi_j \perp U_j.
\end{align*}
Under $\Prob_0 = H_0$ (scale consistency) and $\Prob_{\kappa^*} = H_1$
(scale inconsistency at level $\kappa^*$).

\begin{theorem}[Minimax lower bound]
\label{thm:lecam-lower}
For any test $\phi: \R^L \to \{0,1\}$ (measurable),
\[
    \Prob_0(\phi = 1) + \Prob_{\kappa^*}(\phi = 0)
    \;\geq\;
    1 - \frac{1}{2}\norm{\Prob_0 - \Prob_{\kappa^*}}_{\mathrm{TV}}.
\]
Moreover, for $\kappa^* = c\,(nL)^{-1/2}$ with $c > 0$ sufficiently small,
\[
    \norm{\Prob_0 - \Prob_{\kappa^*}}_{\mathrm{TV}} \;\leq\; \tfrac{1}{2},
\]
so the sum of Type-I and Type-II error probabilities is at least $\frac{3}{4}$
for any test.  In particular, the minimax separation rate is $(nL)^{-1/2}$:
no test is consistent when $\kappa^* = o((nL)^{-1/2})$.
\end{theorem}

\begin{proof}
We apply Le~Cam's two-point method.
The total variation distance between $\Prob_0$ and $\Prob_{\kappa^*}$ is
bounded by Pinsker's inequality:
\[
    \norm{\Prob_0 - \Prob_{\kappa^*}}_{\mathrm{TV}}^2
    \;\leq\; \tfrac{1}{2} D_{\mathrm{KL}}(\Prob_{\kappa^*} \| \Prob_0).
\]
Under $\Prob_\kappa$, the marginal law of $Y_j$ is
$\calN(H\log j,\, \kappa^2 + v_j)$.  Under $\Prob_0$, it is
$\calN(H\log j,\, v_j)$.  Independence across lags gives:
\begin{align}
    D_{\mathrm{KL}}(\Prob_{\kappa^*} \| \Prob_0)
    &= \sum_{j=1}^L D_{\mathrm{KL}}\!\left(\calN(0,\kappa^{*2}+v_j)\|\calN(0,v_j)\right)
    \notag\\
    &= \sum_{j=1}^L \left[\frac{\kappa^{*2}}{2v_j} -
       \frac{1}{2}\log\!\left(1+\frac{\kappa^{*2}}{v_j}\right)\right]
     \notag\\
    &\leq \sum_{j=1}^L \frac{\kappa^{*2}}{2 v_j}
    \;=\; \frac{n\kappa^{*2}}{2} \sum_{j=1}^L j^{2H}.
    \label{eq:kl-bound}
\end{align}
Since $\sum_{j=1}^L j^{2H} \leq L^{2H+1}$ and $\sum_{j=1}^L j^{2H} \geq L$
for $j \geq 1$, we have
\[
    D_{\mathrm{KL}}(\Prob_{\kappa^*}\|\Prob_0) \;\leq\;
    \frac{n\kappa^{*2} L^{2H+1}}{2}.
\]
For $\kappa^* = c(nL)^{-1/2}$,
\[
    D_{\mathrm{KL}} \leq \frac{n \cdot c^2/(nL) \cdot L^{2H+1}}{2}
    = \frac{c^2 L^{2H}}{2}.
\]

\begin{remark}
    The bound in \eqref{eq:kl-bound} shows the \emph{exact} rate-limiting
    term.  For $H < 1/2$, $\sum j^{2H} \asymp L^{2H+1}/(2H+1)$, so
    $D_{\mathrm{KL}} \asymp n\kappa^{*2} L^{2H+1}$.  For the lower bound
    to give $D_{\mathrm{KL}} \leq 1/2$, we need $\kappa^* \lesssim
    (nL^{2H+1})^{-1/2}$, which is the sharper rate in the regime $L \to \infty$.
    For $L$ fixed, both $L^{2H+1}$ and $L$ are constants and the rate
    reduces to $n^{-1/2}$.  The case $L \to \infty$ is developed in
    \Cref{sec:adaptivity}.
\end{remark}

For $L$ fixed and $c$ chosen so that $c^2 L^{2H}/2 \leq 1/8$,
we get $\norm{\Prob_0 - \Prob_{\kappa^*}}_{\mathrm{TV}} \leq 1/2$
and the conclusion follows from Le~Cam's lemma.
\end{proof}

\subsection{Matching upper bound}

The following proposition shows our test achieves the lower bound.

\begin{proposition}[Attainability]
\label{prop:upper-bound}
The test \eqref{eq:test-rule} satisfies: for $\kappa^* = C_1 (nL)^{-1/2}$
with $C_1$ sufficiently large,
\[
    \Prob_0(\text{reject}) \leq \alpha + o(1),\qquad
    \Prob_{\kappa^*}(\text{reject}) \geq 1 - \beta - o(1),
\]
for any $\beta > 0$.  In particular, the test is consistent at the
minimax rate.
\end{proposition}

\begin{proof}
Under $H_1$, the statistic $Q = \sum_j \hat w_j \hat U_j^2$ has
\[
    \E_{\kappa^*}[Q] = (L-2) + \frac{\kappa^{*2}}{1} \sum_j \hat w_j
    \;\approx\; (L-2) + n\kappa^{*2} \sum_j D_j^2/\sigma_0^2.
\]
The non-centrality parameter is $\lambda = n\kappa^{*2}\sum_j j^{2H}/\sigma_0^2
\asymp nL\kappa^{*2}$ for $H$ bounded.  At $\kappa^* = C_1(nL)^{-1/2}$,
$\lambda \asymp C_1^2$.  For $C_1$ large, the non-central $\chi^2(L-2,\lambda)$
distribution places mass near $q_{L-2,1-\alpha} + \lambda$, so the power
approaches 1.
\end{proof}

\section{Adaptivity in $H$}
\label{sec:adaptivity}

In practice, $H$ is unknown.  The FWLS statistic $Q$ depends on $H$ only
through the pilot weights $\hat w_j$; the critical value $q_{L-2,1-\alpha}$
is free of $H$.  We now show this adaptation is cost-free.

\begin{theorem}[Adaptive attainment of minimax rate]
\label{thm:adaptivity}
Suppose $L = L_n \to \infty$ with $L_n = o(n^{1/(2\overline{H}+1)})$.
Under \Cref{ass:signal,ass:sigma,ass:lags} and $H_0$:
\begin{enumerate}[label=(\alph*)]
    \item $Q \dto \chi^2(L-2)$ and the test \eqref{eq:test-rule} has
    asymptotic size $\alpha$.
    \item Under $H_1$ with $\kappa \geq C_1 (nL^{2H+1})^{-1/2}$, the test
    has power tending to 1.
    \item No test achieves power tending to 1 when
    $\kappa = o((nL^{2H+1})^{-1/2})$, for any $H$-adaptive sequence of tests.
\end{enumerate}
\end{theorem}

\begin{proof}[Proof sketch]
\textbf{Parts (a) and (b).}
When $L \to \infty$, the key step is showing that the pilot estimation
error $\hat H_{\mathrm{pil}} - H = O_p((nL^{2H+1}\log^2 L)^{-1/2})$
(oracle OLS rate in log-scale, using the bounds of \Cref{prop:oracle-clt})
translates to a weight perturbation $\delta_j = O_p((nL^{2H+1})^{-1/2}
\log j)$ that contributes negligibly to $Q$:
\begin{align*}
    Q - Q^* &= \sum_j w_j^* \delta_j \hat U_j^2 + O_p\bigl(n^{-1}\log L\bigr) \\
            &= O_p\bigl((nL^{2H+1})^{-1/4} L^{1/2}\bigr) = o_p(1).
\end{align*}
under the assumed growth of $L$.  Convergence of the $\chi^2(L-2)$
distribution then follows from the Berry-Esseen theorem applied to the
standardised statistic $T$.

\textbf{Part (c).}
The lower bound in \Cref{thm:lecam-lower} extends to $L \to \infty$ by
tracking the full sum $\sum_j j^{2H} \asymp L^{2H+1}/(2H+1)$ in
\eqref{eq:kl-bound}, giving the minimax rate $(nL^{2H+1})^{-1/2}$.
Adaptivity in $H$ is free because the critical value $q_{L-2,1-\alpha}$
and the rejection region do not depend on $H$; the test remains valid
for all $H \in [\underline{H}, \overline{H}]$ simultaneously.
\end{proof}

\section{Operational Implication}
\label{sec:application}

The inferred scale law supplies a low-dimensional summary of lagged transport
geometry for downstream memory rules.  In particular, when the power-law
family is not rejected, the estimated pair $(\widehat\zeta, \widehat H)$ can
serve as an input to horizon-based memory laws.  Under
\Cref{ass:signal,ass:sigma,ass:lags}, the adequacy procedure has false alarm
rate at most $\alpha = 5\%$, power tending to one above the boundary
$\kappa^* = C_1(nL)^{-1/2}$, and no tuning of $H$ required.

\section{Conclusion}
\label{sec:conclusion}

We have established a minimax inference theory for power-law lag geometry
under noisy transport estimates.  The low-dimensional family
$D_j = \zeta j^H$ remains inferable despite parameter-coupled noise: the
roughness exponent admits a feasible oracle-equivalent estimator, adequacy of
the family is testable through a residual statistic with an exact oracle null
law, and the same procedure attains the minimax boundary without prior
knowledge of $H$.

The simulations confirm exact calibration, FWLS-oracle agreement, and power
curves matching the theoretical boundary.  The procedure is practical---it
requires only standard WLS computations.

The remaining gaps are limited and explicit:
\begin{itemize}[leftmargin=*]
    \item \emph{Non-Gaussian noise.}  The delta-method approximation
    \eqref{eq:log-noise} holds in generality; the $\chi^2$ limit requires
    only moment conditions, not exact Gaussianity.  A Berry-Esseen rate
    for the $\chi^2(L-2)$ approximation would sharpen the finite-$n$
    guarantees.
    \item \emph{Unknown $\sigma_0$.}  Replacing $\sigma_0$ by a
    consistent plug-in estimator adds an $O_p(n^{-1/2})$ perturbation
    that does not change the asymptotic theory.
    \item \emph{Sequential testing.}  A sequential (anytime-valid)
    version of the test would allow continuous monitoring without inflating
    the false alarm rate.
    \item \emph{Composite $H$ classes.}  Extending the lower bound to
    adaptive testing over H"older classes of the underlying drift (rather
    than the scalar $H$) is a natural next step.
\end{itemize}

\section{Proof of the FWLS Rate Constant}
\label{app:rate-constant}

We compute $[(\mathbf{X}^\top \mathbf{W}^* \mathbf{X})^{-1}]_{22}$
explicitly for weights $w_j^* = n j^{2H}/\sigma_0^2$.

Let $S_k = \sum_{j=1}^L j^{2H} x_j^k$ for $k = 0, 1, 2$ where
$x_j = \log j$.  Then
\[
    \mathbf{X}^\top \mathbf{W}^* \mathbf{X}
    = \frac{n}{\sigma_0^2} \begin{pmatrix} S_0 & S_1 \\ S_1 & S_2 \end{pmatrix},
\]
and the $(2,2)$ entry of the inverse is
\[
    \bigl[(\mathbf{X}^\top \mathbf{W}^* \mathbf{X})^{-1}\bigr]_{22}
    = \frac{\sigma_0^2}{n} \cdot \frac{S_0}{S_0 S_2 - S_1^2}.
\]
For $L = 20$, $H = 0.6$:
$S_0 \approx 166.2$, $S_1 \approx 347.4$, $S_2 \approx 836.7$,
$S_0 S_2 - S_1^2 \approx 166.2 \times 836.7 - 347.4^2 \approx 18219$,
giving
$\Var(\hat H^*) \approx \sigma_0^2 \times 166.2 / (n \times 18219)
\approx 9.13 \times 10^{-3} \sigma_0^2 / n$.
Thus $\mathrm{RMSE}(\hat H^*) \approx \sqrt{9.13 \times 10^{-3}/n}
\cdot \sigma_0$.
Our theoretical rate factor is $(n L^{2H+1})^{-1/2} = (n \times 20^{2.2})^{-1/2}
\approx (n \times 70.4)^{-1/2}$.
The ratio gives $C = \sqrt{9.13 \times 10^{-3} \times 70.4} \approx 0.802$
for the oracle.  The FWLS adds a constant factor (verified to be $\approx 3.22$
in \Cref{tab:rate}); the discrepancy with the oracle bound reflects the
additional variance from the pilot step and the delta-method linearization
error.

\section{Auxiliary Lemmas}
\label{app:lemmas}

\begin{lemma}[KL divergence between scaled Gaussians]
\label{lem:kl-gaussian}
For $\kappa^2 \leq v$,
\begin{align*}
    D_{\mathrm{KL}}\bigl(\calN(0, \kappa^2+v)\|\calN(0,v)\bigr)
    &= \frac{1}{2}\left[\frac{\kappa^2}{v} - \log\!\left(1 + \frac{\kappa^2}{v}\right)\right] \\
    &\leq \frac{\kappa^4}{4v^2}.
\end{align*}
\end{lemma}

\begin{proof}
Standard formula; the upper bound is the second-order Taylor remainder of
$-\log(1+x)$ at $x = \kappa^2/v$.
\end{proof}

\begin{lemma}[OLS rate in log-scale regression]
\label{lem:ols-rate}
In the pilot OLS regression of $Y_j = \alpha + H x_j + U_j$ with
$U_j \iid \calN(0, v)$ (ignoring heteroscedasticity), the OLS estimator
$\hat H_{\mathrm{pil}}$ satisfies
\[
    \hat H_{\mathrm{pil}} - H = \frac{\sum_j (x_j - \bar x) U_j}
    {\sum_j (x_j - \bar x)^2}
    \;\sim\;
    \calN\!\left(0,\; \frac{v}{\sum_j (x_j-\bar x)^2}\right).
\]
For $L$ fixed and $v = \sigma_0^2 n^{-1} j^{-2H}$ heteroscedastic,
$\hat H_{\mathrm{pil}} - H = O_p(n^{-1/2})$.
\end{lemma}

\begin{proof}
Follows from the Gauss-Markov structure of OLS and the boundedness of
$v_j \leq \sigma_0^2/n$ (since $j \geq 1$, $j^{2H} \geq 1$).
\end{proof}
