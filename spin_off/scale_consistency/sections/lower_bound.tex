\section{Lower Bound}
\label{sec:lower}

We now show that no test can do better than the information-scale boundary
generated by the weighted lag-energy.

\subsection{Gaussian Variance-Perturbation Model}

Fix $H \in (0,1)$, $\zeta = 1$, $\sigma_0 = 1$.  For $\kappa \geq 0$,
define the probability measure $\Prob_\kappa$ on $\R^L$ under which
\begin{align*}
    Y_j &= H\log j + \kappa\,\xi_j + U_j,\\
    \xi_j &\iid \calN(0,1), \qquad U_j \sim \calN(0, v_j),\\
    v_j &= n^{-1} j^{-2H}, \qquad \xi_j \perp U_j.
\end{align*}
Under $\Prob_0 = H_0$ (membership in the power-law family) and
$\Prob_{\kappa^*} = H_1$ (power-law departure at level $\kappa^*$).

\begin{theorem}[Minimax Separation Lower Bound]
\label{thm:lecam-lower}
For any test $\phi: \R^L \to \{0,1\}$ (measurable),
\[
    \Prob_0(\phi = 1) + \Prob_{\kappa^*}(\phi = 0)
    \;\geq\;
    1 - \frac{1}{2}\norm{\Prob_0 - \Prob_{\kappa^*}}_{\mathrm{TV}}.
\]
Moreover, if
\[
    \kappa^* = c\left(n\sum_{j=1}^L j^{2H}\right)^{-1/2}
\]
with $c > 0$ sufficiently small, then
\[
    \norm{\Prob_0 - \Prob_{\kappa^*}}_{\mathrm{TV}} \;\leq\; \tfrac{1}{2},
\]
so the sum of Type-I and Type-II error probabilities is at least $\frac{3}{4}$
for any test.  In particular, the minimax separation boundary is
\[
    \kappa^* \asymp \left(n\sum_{j=1}^L j^{2H}\right)^{-1/2}.
\]
For fixed $L$, this reduces to $n^{-1/2}$ up to constants.  For consecutive
lags $1,\dots,L_n$ with $L_n \to \infty$, it becomes
$(nL_n^{2H+1})^{-1/2}$ up to constants.
\end{theorem}

\begin{proof}
We apply Le~Cam's two-point method \citep{lecam1973convergence}.
The total variation distance between $\Prob_0$ and $\Prob_{\kappa^*}$ is
bounded by Pinsker's inequality:
\[
    \norm{\Prob_0 - \Prob_{\kappa^*}}_{\mathrm{TV}}^2
    \;\leq\; \tfrac{1}{2} D_{\mathrm{KL}}(\Prob_{\kappa^*} \| \Prob_0).
\]
Under $\Prob_\kappa$, the marginal law of $Y_j$ is
$\calN(H\log j,\, \kappa^2 + v_j)$.  Under $\Prob_0$, it is
$\calN(H\log j,\, v_j)$.  Independence across lags gives:
\begin{align}
    D_{\mathrm{KL}}(\Prob_{\kappa^*} \| \Prob_0)
    &= \sum_{j=1}^L D_{\mathrm{KL}}\!\left(\calN(0,\kappa^{*2}+v_j)\|\calN(0,v_j)\right)
    \notag\\
    &= \sum_{j=1}^L \left[\frac{\kappa^{*2}}{2v_j} -
       \frac{1}{2}\log\!\left(1+\frac{\kappa^{*2}}{v_j}\right)\right]
     \notag\\
    &\leq \sum_{j=1}^L \frac{\kappa^{*2}}{2 v_j}
    \;=\; \frac{n\kappa^{*2}}{2} \sum_{j=1}^L j^{2H}.
    \label{eq:kl-bound}
\end{align}
For the displayed choice of $\kappa^*$,
\[
    D_{\mathrm{KL}}(\Prob_{\kappa^*}\|\Prob_0)
    \leq \frac{c^2}{2}.
\]

\begin{remark}
    The bound in \eqref{eq:kl-bound} shows the exact rate-limiting term:
    the weighted lag-energy $\sum_{j=1}^L j^{2H}$.  For consecutive lags,
    $\sum_{j=1}^L j^{2H} \asymp L^{2H+1}/(2H+1)$, so the same KL bound yields
    the growing-$L$ boundary $(nL^{2H+1})^{-1/2}$.
\end{remark}

Choosing $c^2 \leq 1/4$, we get
$\norm{\Prob_0 - \Prob_{\kappa^*}}_{\mathrm{TV}} \leq 1/2$
and the conclusion follows from Le~Cam's lemma.
\end{proof}

\begin{remark}[Two asymptotic regimes]
    The lower bound is controlled by the same information law in both regimes.
    For fixed $L$, the lag-energy $\sum_{j=1}^L j^{2H}$ is a constant, so the
    separation boundary is parametric in $n$.  For consecutive lags with
    $L=L_n \to \infty$, the sum is asymptotic to $L_n^{2H+1}/(2H+1)$, so the
    boundary becomes $(nL_n^{2H+1})^{-1/2}$.
\end{remark}

\begin{remark}[Fixed-$L$ boundary response]
    For fixed $L$, the information-scale boundary is a local-testing scale, not
    a consistency regime.  Under departures of the form
    $\kappa = c\bigl(n\sum_{j=1}^L j^{2H}\bigr)^{-1/2}$, the adequacy test is
    indexed by the constant $c$: larger $c$ produces larger variance inflation
    in the residual quadratic form, but power does not converge to one merely by
    sending $n \to \infty$ with $L$ fixed.  The theorem-level consistency
    statement belongs to the growing-lag regime of \Cref{sec:adaptivity}.
\end{remark}
